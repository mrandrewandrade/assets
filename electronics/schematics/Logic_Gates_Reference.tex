\newcommand{\twogate}[3]{%
  \node[#1,twogate] (#2) at #3 {};
  \draw (#2.input 1) -- ++(-0.65,0) node[left,pinlabel] {$A$};
  \draw (#2.input 2) -- ++(-0.65,0) node[left,pinlabel] {$B$};
  \draw (#2.output) -- ++(0.65,0) node[right,pinlabel] {$Y$};
}
